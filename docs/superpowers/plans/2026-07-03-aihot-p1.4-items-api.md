# P1.4 Items API (`/api/public/items`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the public `GET /api/public/items` endpoint on top of P1.1's `items.Store`, implementing the openapi contract's API-layer concerns the store deliberately left out: opaque cursor encode/decode, the 7-day `since` clamp, `mode` (selected/all), query validation, the public `Item` projection (stripping internal fields), the `ItemList` envelope (`count`/`hasNext`/`nextCursor`), and weak ETag / 304.

**Architecture:** A `PublicItem` DTO + `toPublic` projection expose only browser-visible fields. A `cursor` codec encodes the typed `items.Cursor` into an opaque base64 token and decodes it leniently (invalid/corrupt → nil → first page, per contract). A `params` parser validates and normalizes query params against the contract (take 1–100/default 50, since ≤7d with clamp + future→400, q len 2–200, mode, category). A `Handler` wires store→projection→envelope, computes a weak ETag over the response, and honors `If-None-Match` with 304. Mounted via the same `AddHTTPHandle` escape hatch P1.0 used (no IDL/nuwa-gen).

**Tech Stack:** Go (module `aihot-server`), stdlib `net/http`, the existing `internal/items` store. Tests: pure unit for cursor/projection/params, `httptest` for the handler, live Melos PG for end-to-end.

## Contract recap (from `docs/references/openapi.yaml`, `/api/public/items`)

- Order by `publishedAt` desc (the store already does `COALESCE(published_at,epoch) DESC, id DESC`).
- `mode`: `selected` (default) | `all`. selected → `Selected=true`. (The store's always-on `present=true AND duplicate_of_id IS NULL` covers the implicit filters; the mode=all aiRelevance<60 / arxiv / mp_hot policy is NOT in P1.4 scope — see Self-Review.)
- `category`: one of the 5 slugs, else 400.
- `since`: ISO datetime, within last 7 days. Unset → `now-7d`. Older than 7d → clamp to `now-7d`. Future (> now+1min) → 400.
- `take`: 1–100, default 50; out of range / non-integer → 400.
- `cursor`: opaque token = a prior `nextCursor`. Invalid/corrupt/expired → silently return first page (NOT an error).
- `q`: length 2–200; <2 chars → treated as no search; >200 → truncate to 200.
- Response `ItemList`: `{count, hasNext, nextCursor, items[]}`. `items[]` are the public `Item` shape (id, title, title_en, url, permalink, source, publishedAt, summary, category, score, selected) — internal fields (ai_relevance, ai_selected, cluster_id, duplicate_of_id, present, body, timeline_at) are NOT exposed.
- Weak ETag `W/"items-<hash>"` over the whole page; `If-None-Match` match → 304 empty body. `Cache-Control: public, s-maxage=300, stale-while-revalidate=300`.
- Errors: 400 `{"error":"<reason>"}`.

package publicapi

import (
	"encoding/base64"
	"encoding/json"
	"time"

	"aihot-server/internal/items"
)

// cursorPayload is the internal, NON-CONTRACT wire form of a cursor. The encoding
// is opaque to clients and may change without notice.
type cursorPayload struct {
	S int64  `json:"s"` // sort key, unix micro
	I string `json:"i"` // id
}

// encodeCursor turns a typed cursor into an opaque base64url token. Nil → "".
func encodeCursor(c *items.Cursor) string {
	if c == nil {
		return ""
	}
	b, err := json.Marshal(cursorPayload{S: c.SortKey.UnixMicro(), I: c.ID})
	if err != nil {
		return ""
	}
	return base64.RawURLEncoding.EncodeToString(b)
}

// decodeCursor is lenient: any malformed / empty / incomplete token returns nil,
// which the caller treats as "no cursor → first page" (per the opaque-token contract).
func decodeCursor(tok string) *items.Cursor {
	if tok == "" {
		return nil
	}
	b, err := base64.RawURLEncoding.DecodeString(tok)
	if err != nil {
		return nil
	}
	var p cursorPayload
	if err := json.Unmarshal(b, &p); err != nil {
		return nil
	}
	if p.I == "" {
		return nil
	}
	return &items.Cursor{SortKey: time.UnixMicro(p.S).UTC(), ID: p.I}
}
```

package publicapi

import (
	"errors"
	"net/url"
	"strconv"
	"time"

	"aihot-server/internal/items"
)

// errBadRequest signals a 400 to the handler.
var errBadRequest = errors.New("bad request")

const (
	defaultTake  = 50
	maxTake      = 100
	sinceWindow  = 7 * 24 * time.Hour
	futureSkew   = time.Minute
	maxQ         = 200
)

var apiCategories = map[string]bool{
	items.CategoryAIModels:   true,
	items.CategoryAIProducts: true,
	items.CategoryIndustry:   true,
	items.CategoryPaper:      true,
	items.CategoryTip:        true,
}

// parseListParams validates the query against the openapi contract and returns
// the store-level ListParams. `now` is injected for testability. Returns
// errBadRequest on any contract violation (→ 400).
func parseListParams(q url.Values, now time.Time) (items.ListParams, error) {
	var p items.ListParams

	// mode: default selected → Selected=true; all → no constraint.
	switch q.Get("mode") {
	case "", "selected":
		tru := true
		p.Selected = &tru
	case "all":
		// leave nil
	default:
		return p, errBadRequest
	}

	// category (optional; must be a known slug).
	if c := q.Get("category"); c != "" {
		if !apiCategories[c] {
			return p, errBadRequest
		}
		cc := c
		p.Category = &cc
	}

	// take: 1..100, default 50.
	p.Limit = defaultTake
	if t := q.Get("take"); t != "" {
		n, err := strconv.Atoi(t)
		if err != nil || n < 1 || n > maxTake {
			return p, errBadRequest
		}
		p.Limit = n
	}

	// since: within last 7d. unset → now-7d. older → clamp. future(>now+1min) → 400.
	lower := now.Add(-sinceWindow)
	if s := q.Get("since"); s != "" {
		ts, err := time.Parse(time.RFC3339, s)
		if err != nil {
			return p, errBadRequest
		}
		if ts.After(now.Add(futureSkew)) {
			return p, errBadRequest
		}
		if ts.Before(lower) {
			ts = lower
		}
		tt := ts.UTC()
		p.Since = &tt
	} else {
		p.Since = &lower
	}

	// q: <2 chars ignored; >200 truncated.
	if qs := q.Get("q"); len([]rune(qs)) >= 2 {
		if len([]rune(qs)) > maxQ {
			qs = string([]rune(qs)[:maxQ])
		}
		p.Q = &qs
	}

	// cursor: opaque; invalid → nil After (first page), never an error.
	if cur := q.Get("cursor"); cur != "" {
		p.After = decodeCursor(cur)
	}

	return p, nil
}
```

package publicapi

import (
	"context"
	"crypto/sha1"
	"encoding/json"
	"fmt"
	"net/http"
	"time"

	"aihot-server/internal/items"
)

// Lister is the store surface the handler needs (satisfied by *items.Store).
type Lister interface {
	List(ctx context.Context, p items.ListParams) ([]items.Item, error)
}

// Clock returns the current time (injected for testability).
type Clock func() time.Time

// ItemsHandler serves GET /api/public/items.
type ItemsHandler struct {
	store Lister
	now   Clock
}

func NewItemsHandler(store Lister, now Clock) *ItemsHandler {
	return &ItemsHandler{store: store, now: now}
}

func (h *ItemsHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	p, err := parseListParams(r.URL.Query(), h.now())
	if err != nil {
		writeError(w, http.StatusBadRequest, "invalid query parameters")
		return
	}

	// Request one extra row to detect whether a further page exists.
	take := p.Limit
	p.Limit = take + 1

	rows, err := h.store.List(r.Context(), p)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "internal error")
		return
	}

	hasNext := len(rows) > take
	if hasNext {
		rows = rows[:take]
	}

	env := ItemList{
		Count: len(rows),
		Items: make([]PublicItem, 0, len(rows)),
	}
	for _, it := range rows {
		env.Items = append(env.Items, toPublic(it))
	}
	env.HasNext = hasNext
	if hasNext && len(rows) > 0 {
		last := rows[len(rows)-1]
		tok := encodeCursor(&items.Cursor{SortKey: last.SortKey(), ID: last.ID})
		env.NextCursor = &tok
	}

	body, err := json.Marshal(env)
	if err != nil {
		writeError(w, http.StatusInternalServerError, "encoding error")
		return
	}

	etag := weakETag(body)
	w.Header().Set("ETag", etag)
	w.Header().Set("Cache-Control", "public, s-maxage=300, stale-while-revalidate=300")
	if match := r.Header.Get("If-None-Match"); match == etag {
		w.WriteHeader(http.StatusNotModified)
		return
	}
	w.Header().Set("Content-Type", "application/json")
	_, _ = w.Write(body)
}

func weakETag(body []byte) string {
	sum := sha1.Sum(body)
	return fmt.Sprintf(`W/"items-%x"`, sum[:8])
}

func writeError(w http.ResponseWriter, code int, msg string) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(code)
	_ = json.NewEncoder(w).Encode(map[string]string{"error": msg})
}
